   function ref_level_name(code) result(name)
      !! `RDB_REF_LEVEL_*` code -> the routine's `ref_level` string ("" if bad).
      integer, intent(in) :: code
      character(len=11) :: name
      select case (code)
      case (1)
         name = "keep_h"
      case (2)
         name = "bottom_rest"
      case default
         name = ""
      end select
   end function ref_level_name

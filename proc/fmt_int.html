   pure function fmt_int(v) result(str)
      !! Format an integer.
      integer, intent(in) :: v
      character(len=:), allocatable :: str
      character(len=32) :: buf
      write (buf, "(I0)") v
      str = trim(adjustl(buf))
   end function fmt_int

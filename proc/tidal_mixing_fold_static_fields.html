public pure subroutine tidal_mixing_fold_static_fields(this, nx, ny, depth, wet)

Fold the setup-time masks and clamps into the static 2-D fields. Call once on the host after tideamp and h2 hold the raw (file or uniform) values and BEFORE enter_data. Per cell, with D the resting column depth: * U = tideamp where wet and D >= min_zbot, else 0; * <h^2> <- min(<h^2>, (frac_rough*D)^2), 0 on dry cells (Jayne & St Laurent 2001 roughness cap); * c_e = 0.5*rho0*kappa_h2*kappa_itides*<h^2>*U^2 [kg s-2], so the energy input is E = min(c_e*N_b, e_max).

   pure subroutine tidal_mixing_fold_static_fields(this, nx, ny, depth, wet)
      !! Fold the setup-time masks and clamps into the static 2-D fields.
      !! Call once on the host after `tideamp` and `h2` hold the raw
      !! (file or uniform) values and BEFORE `enter_data`.  Per cell, with
      !! `D` the resting column depth:
      !!   * `U = tideamp` where wet and `D >= min_zbot`, else 0;
      !!   * `<h^2> <- min(<h^2>, (frac_rough*D)^2)`, 0 on dry cells
      !!     (Jayne & St Laurent 2001 roughness cap);
      !!   * `c_e = 0.5*rho0*kappa_h2*kappa_itides*<h^2>*U^2`  [kg s-2],
      !!     so the energy input is `E = min(c_e*N_b, e_max)`.
      type(ocean_tidal_mixing_t), intent(inout) :: this
      integer, intent(in) :: nx, ny
      real(wp), intent(in) :: depth(nx, ny)
         !! Column depth D (m), positive.  Setup passes the layer-thickness
         !! sum `sum(h_layer)`, which equals the resting depth on a cold
         !! start but INCLUDES the free-surface displacement eta on a warm
         !! restart (so `min_zbot` and the `frac_rough*D` cap see D + eta).
      real(wp), intent(in) :: wet(nx, ny)
         !! Wet mask (1 wet, 0 land).
      integer :: i, j
      real(wp) :: dd, h2c, u

      do j = 1, ny
         do i = 1, nx
            dd = max(depth(i, j), 0.0_wp)
            u = this%tideamp(i, j)
            h2c = this%h2(i, j)
            if (wet(i, j) <= 0.0_wp .or. dd < this%min_zbot) then
               u = 0.0_wp
               h2c = 0.0_wp
            end if
            h2c = min(h2c, (this%frac_rough*dd)**2)
            this%tideamp(i, j) = u
            this%h2(i, j) = h2c
            this%c_e(i, j) = 0.5_wp*this%rho0*this%kappa_h2*this%kappa_itides*h2c*u**2
         end do
      end do
   end subroutine tidal_mixing_fold_static_fields

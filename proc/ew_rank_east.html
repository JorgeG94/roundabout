   pure function ew_rank_east() result(r)
      !! East neighbour rank with periodic wrap-around.
      integer :: r
      integer :: rx_e
      rx_e = oh_decomp%rx + 1
      if (rx_e >= oh_decomp%px) rx_e = 0
      r = decomp_rank_from_coords(oh_decomp%px, rx_e, oh_decomp%ry)
   end function ew_rank_east

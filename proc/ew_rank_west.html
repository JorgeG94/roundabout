   pure function ew_rank_west() result(r)
      !! West neighbour rank with periodic wrap-around.
      integer :: r
      integer :: rx_w
      rx_w = oh_decomp%rx - 1
      if (rx_w < 0) rx_w = oh_decomp%px - 1
      r = decomp_rank_from_coords(oh_decomp%px, rx_w, oh_decomp%ry)
   end function ew_rank_west

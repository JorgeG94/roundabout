   pure function is_repeat_count(word) result(yes)
      !! True if `word` looks like `N*value` (integer, then '*').
      character(len=*), intent(in) :: word
      logical :: yes
      integer :: star, i
      character :: c
      yes = .false.
      star = index(word, "*")
      if (star <= 1) return
      do i = 1, star - 1
         c = word(i:i)
         if (c < "0" .or. c > "9") return
      end do
      yes = .true.
   end function is_repeat_count

   pure subroutine lower_median(a, n, m)
      !! Lower median `m` of `a(1:n)` by quickselect.  `a` is a scratch buffer:
      !! it is partially reordered.
      integer, intent(in) :: n
      real(wp), intent(inout) :: a(:)
      real(wp), intent(out) :: m
      integer :: lo, hi, kth, i, j
      real(wp) :: pivot, t

      kth = (n + 1)/2
      lo = 1
      hi = n
      do while (lo < hi)
         pivot = a((lo + hi)/2)
         i = lo
         j = hi
         do while (i <= j)
            do while (a(i) < pivot)
               i = i + 1
            end do
            do while (a(j) > pivot)
               j = j - 1
            end do
            if (i <= j) then
               t = a(i)
               a(i) = a(j)
               a(j) = t
               i = i + 1
               j = j - 1
            end if
         end do
         if (kth <= j) then
            hi = j
         else if (kth >= i) then
            lo = i
         else
            exit
         end if
      end do
      m = a(kth)
   end subroutine lower_median

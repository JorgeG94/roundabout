   pure function ns_rank_south() result(r)
      !! South neighbour rank with periodic wrap-around.
      integer :: r
      integer :: ry_s
      ry_s = oh_decomp%ry - 1
      if (ry_s < 0) ry_s = oh_decomp%py - 1
      r = decomp_rank_from_coords(oh_decomp%px, oh_decomp%rx, ry_s)
   end function ns_rank_south

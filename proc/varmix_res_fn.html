varmix_res_fn

private pure function varmix_res_fn(f2_dx2, beta_dx2, cg1, alpha, p) result(r)

Divide-free resolution function for integer power p >= 1 (Hallberg 2013, Ocean Modelling 72): with s = sqrt(dx_term), Res_fn = s^p/(s^p + (alpha*cg1)^p), where dx_term = f2_dx2 + cg1*beta_dx2 is a squared velocity. p=2 reduces to dx_term/(dx_term + (alpha*cg1)^2); even p uses dx_term^(p/2) (exact, no sqrt); odd p uses sqrt(dx_term)^p. -> 1 where unresolved, -> 0 where Ld>>dx.

   pure function varmix_res_fn(f2_dx2, beta_dx2, cg1, alpha, p) result(r)
      !$acc routine seq
      !! Divide-free resolution function for integer power `p >= 1`
      !! (Hallberg 2013, Ocean Modelling 72): with `s = sqrt(dx_term)`,
      !! `Res_fn = s^p/(s^p + (alpha*cg1)^p)`, where `dx_term =
      !! f2_dx2 + cg1*beta_dx2` is a squared velocity.  p=2 reduces to
      !! `dx_term/(dx_term + (alpha*cg1)^2)`; even p uses `dx_term^(p/2)` (exact,
      !! no sqrt); odd p uses `sqrt(dx_term)^p`.  -> 1 where unresolved,
      !! -> 0 where Ld>>dx.
      real(wp), intent(in) :: f2_dx2, beta_dx2, cg1, alpha
      integer, intent(in) :: p
      real(wp) :: r, dx_term, num, den_add
      integer :: ph
      dx_term = f2_dx2 + cg1*beta_dx2
      if (p == 2) then
         num = dx_term
         den_add = (alpha*cg1)*(alpha*cg1)
      else if (mod(p, 2) == 0) then
         ph = p/2
         num = dx_term**ph
         den_add = (alpha*cg1)**p
      else
         num = sqrt(max(dx_term, 0.0_wp))**p
         den_add = (alpha*cg1)**p
      end if
      r = num/(num + den_add)
   end function varmix_res_fn

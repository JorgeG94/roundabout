subroutine validate_balanced_vortex(cfg, has_error)

Config conflicts of ic_config = "balanced_vortex". The IC is built from an analytic Boussinesq hydrostatic pressure anomaly, so it needs a linear EOS (the density anomaly is -alpha_T * dT), a cartesian beta-plane f, a flat bed, and a vortex that the grid resolves (R >= 3 dx). Gradient-wind real-root existence depends on the field and is checked at seed time.

   subroutine validate_balanced_vortex(cfg, has_error)
      !! Config conflicts of `ic_config = "balanced_vortex"`.  The IC is
      !! built from an analytic Boussinesq hydrostatic pressure anomaly, so
      !! it needs a linear EOS (the density anomaly is `-alpha_T * dT`), a
      !! cartesian beta-plane f, a flat bed, and a vortex that the grid
      !! resolves (R >= 3 dx).  Gradient-wind real-root existence depends on
      !! the field and is checked at seed time.
      type(config_t), intent(in) :: cfg
      logical, intent(inout) :: has_error
      real(wp) :: dmax, dtdz_min, y_lo, y_hi, f_lo, f_hi

      if (trim(cfg%ocean%eos%eos) /= "linear") then
         call logger%error("ic_config='balanced_vortex' requires &ocean_eos_nml "// &
                           "eos='linear' (the analytic pressure anomaly uses "// &
                           "rho' = -alpha_T*dT); got '"//trim(cfg%ocean%eos%eos)//"'")
         has_error = .true.
      end if
      if (trim(cfg%ocean%grid%grid_config) /= "cartesian") then
         call logger%error("ic_config='balanced_vortex' requires &ocean_grid_nml "// &
                           "grid_config='cartesian'")
         has_error = .true.
      end if
      if (trim(cfg%ocean%grid%coriolis_scheme) /= "beta_plane") then
         call logger%error("ic_config='balanced_vortex' requires "// &
                           "coriolis_scheme='beta_plane' (f0 + beta*(y-y_ref))")
         has_error = .true.
      end if
      if (trim(cfg%ocean%topo%topo_config) /= "flat") then
         call logger%error("ic_config='balanced_vortex' requires topo_config='flat'")
         has_error = .true.
      end if
      if (abs(cfg%coriolis_f) < VORTEX_F_MIN) then
         call logger%error("ic_config='balanced_vortex' requires coriolis_f /= 0 "// &
                           "(|f0| >= "//to_string(VORTEX_F_MIN)//" 1/s): geostrophy is "// &
                           "singular on an f = 0 grid")
         has_error = .true.
      end if
      ! f = f0 + beta (y - y_ref) is linear in y, so its minimum |f| over the
      ! whole grid (ghost rows included: y from -nghost*dy to (ny+nghost)*dy)
      ! is at an end or at the zero crossing the seed would divide by.
      y_lo = -real(cfg%nghost, wp)*cfg%dy
      y_hi = (real(cfg%ny, wp) + real(cfg%nghost, wp))*cfg%dy
      f_lo = cfg%coriolis_f + cfg%ocean%topo%coriolis_beta*(y_lo - cfg%ocean%topo%coriolis_y_ref)
      f_hi = cfg%coriolis_f + cfg%ocean%topo%coriolis_beta*(y_hi - cfg%ocean%topo%coriolis_y_ref)
      if (f_lo*f_hi <= 0.0_wp .and. cfg%ocean%topo%coriolis_beta /= 0.0_wp) then
         call logger%error("ic_config='balanced_vortex': f changes sign on the grid; f = 0 at y = "// &
                           to_string(cfg%ocean%topo%coriolis_y_ref - &
                                     cfg%coriolis_f/cfg%ocean%topo%coriolis_beta)// &
                           " m (grid spans y = "//to_string(y_lo)//" .. "//to_string(y_hi)// &
                           " m); geostrophy is singular there")
         has_error = .true.
      else if (min(abs(f_lo), abs(f_hi)) < VORTEX_F_MIN) then
         call logger%error("ic_config='balanced_vortex': |f| falls below "// &
                           to_string(VORTEX_F_MIN)//" 1/s at the grid edge (f = "// &
                           to_string(f_lo)//" at y = "//to_string(y_lo)//", "// &
                           to_string(f_hi)//" at y = "//to_string(y_hi)//" m)")
         has_error = .true.
      end if
      dmax = max(cfg%dx, cfg%dy)
      if (cfg%ocean%ic%vortex_radius < VORTEX_MIN_CELLS_PER_RADIUS*dmax) then
         call logger%error("vortex_radius = "//to_string(cfg%ocean%ic%vortex_radius)// &
                           " m must be >= "//to_string(VORTEX_MIN_CELLS_PER_RADIUS)// &
                           " dx = "//to_string(VORTEX_MIN_CELLS_PER_RADIUS*dmax)//" m")
         has_error = .true.
      end if
      if (cfg%ocean%ic%vortex_hv <= 0.0_wp) then
         call logger%error("vortex_hv must be > 0")
         has_error = .true.
      end if
      ! Static stability.  Warm core: the anomaly adds stratification.  Cold
      ! core (dT < 0): the anomaly -|dT| exp(-d^2/Hv^2) warms with depth, with
      ! largest gradient at d = Hv/sqrt(2) of sqrt(2) exp(-1/2) |dT|/Hv =
      ! 0.858 |dT|/Hv (K/m); the background dT/dz must exceed it or N^2 < 0.
      if (cfg%ocean%ic%vortex_dT < 0.0_wp .and. cfg%ocean%ic%vortex_hv > 0.0_wp) then
         dtdz_min = sqrt(2.0_wp)*exp(-0.5_wp)*abs(cfg%ocean%ic%vortex_dT)/cfg%ocean%ic%vortex_hv
         if (cfg%ocean%ic%vortex_dT_dz <= dtdz_min) then
            call logger%error("balanced_vortex is statically UNSTABLE (N^2 < 0 in the core): "// &
                              "a cold core dT = "//to_string(cfg%ocean%ic%vortex_dT)// &
                              " degC needs vortex_dT_dz > sqrt(2)*exp(-1/2)*|dT|/Hv = "// &
                              to_string(dtdz_min)//" K/m; got "// &
                              to_string(cfg%ocean%ic%vortex_dT_dz))
            has_error = .true.
         end if
      end if
      if (trim(cfg%ocean%ic%vortex_balance) /= "geostrophic" .and. &
          trim(cfg%ocean%ic%vortex_balance) /= "gradient" .and. &
          trim(cfg%ocean%ic%vortex_balance) /= "none") then
         call logger%error("vortex_balance must be 'geostrophic', 'gradient', or 'none'")
         has_error = .true.
      end if
   end subroutine validate_balanced_vortex

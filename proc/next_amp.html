private subroutine next_amp(lines, n_lines, li, col)

Advance (li,col) to the character AFTER the next ‘&’. On EOF sets li = n_lines + 1. Bare non-blank tokens before ‘&’ are tolerated as inter-group whitespace.

   subroutine next_amp(lines, n_lines, li, col)
      !! Advance (li,col) to the character AFTER the next '&'.  On EOF
      !! sets li = n_lines + 1.  Bare non-blank tokens before '&' are
      !! tolerated as inter-group whitespace.
      character(len=*), intent(in) :: lines(:)
      integer, intent(in) :: n_lines
      integer, intent(inout) :: li, col
      character(len=:), allocatable :: s
      integer :: i

      do while (li <= n_lines)
         s = strip_comment(lines(li))
         do i = col, len(s)
            if (s(i:i) == "&") then
               col = i + 1
               return
            end if
         end do
         li = li + 1
         col = 1
      end do
   end subroutine next_amp

apply_surface_restore_2d_impl

private pure subroutine apply_surface_restore_2d_impl(c, budget, h_layer, wet_mask, dt_piston, tgt, h_min, k_top, nz, nx, ny)

Stamp the per-step restoring increment dt·p·(target - surf)· wet_mask onto the first live layer from the top (k_top, normally nz) of a tracer’s c array and mirror it into the matching budget contributor. Explicit-shape dummies so NVHPC stdpar compiles device kernels against static bounds.

dt_piston = dt·piston [m] is the column-invariant multiplier the caller derives from sf (the rho0·cp cancels — see the caller doc). The surface concentration is the stored c(nz); the content increment enters through rdb_conc_add_source.

   pure subroutine apply_surface_restore_2d_impl(c, budget, h_layer, &
                                                 wet_mask, dt_piston, tgt, &
                                                 h_min, k_top, nz, nx, ny)
      !! Stamp the per-step restoring increment `dt·p·(target - surf)·
      !! wet_mask` onto the first live layer from the top (`k_top`, normally `nz`) of a tracer's c array
      !! and mirror it into the matching budget contributor.
      !! Explicit-shape dummies so NVHPC stdpar compiles device kernels
      !! against static bounds.
      !!
      !! `dt_piston` = dt·piston [m] is the column-invariant multiplier
      !! the caller derives from `sf` (the `rho0·cp` cancels — see the
      !! caller doc).  The surface concentration is the stored `c(nz)`;
      !! the content increment enters through `rdb_conc_add_source`.
      integer, intent(in)    :: nz, nx, ny
      real(wp), intent(inout) :: c(nx, ny, nz)
      real(wp), intent(inout) :: budget(nx, ny, nz)
      real(wp), intent(in)    :: h_layer(nx, ny, nz)
      real(wp), intent(in)    :: wet_mask(nx, ny)
      real(wp), intent(in)    :: dt_piston, tgt, h_min
      integer, intent(in)    :: k_top(nx, ny)
         !! `ms%k_top`; the surface layer is the first LIVE one at or below it.
      integer :: i, j, kt
      real(wp) :: surf, inc
      do concurrent(j=1:ny, i=1:nx) local(surf, inc, kt)
         kt = rdb_vl_first_live(nx, ny, nz, h_layer, i, j, k_top(i, j))
         surf = c(i, j, kt)
         inc = dt_piston*(tgt - surf)*wet_mask(i, j)
         c(i, j, kt) = rdb_conc_add_source(c(i, j, kt), h_layer(i, j, kt), inc)
         budget(i, j, kt) = budget(i, j, kt) + rdb_conc_accepted_source(h_layer(i, j, kt), inc)
      end do
   end subroutine apply_surface_restore_2d_impl

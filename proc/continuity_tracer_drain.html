public subroutine continuity_tracer_drain(grid, metrics, this, ms, ratio, bc, post_weight)

Phase-2 (6b) windowed horizontal tracer-advection drain.

Spends the accumulated face transports this%uhtr/vhtr (built over ratio outer steps in TR_MODE_ACCUMULATE) onto the frozen tracer mass ms%tracers(:)%c using a fixed-budget Colella-Woodward swept-average PPM sub-cycle (Adcroft & Hallberg 2006; MOM6 ADVECT_PPM). Conservation is structural:

hprev = areaT·h_end + div(uhtr) (≈ window-start thickness) Tr_start = c (held) (window-start concentration)

the drain evolves hprev → h_end while moving tracer with each limited transport portion; at the end hprev == h_end so Σ(Tr·hprev) = Σ(Tr_start·hprev_start) to round-off. The concentration is the held c, which is the window-start value.

Pass cap: the sub-cycle runs at most 2*ratio+1 passes, which can leave transport undrained. That leftover is carried into the next window (uhtr/vhtr seeded from it, c untouched) and counted in drain_carry_windows / drain_carry_faces. Known gaps: the carry is not in the restart file (a restart taken while a carry is pending drops it); an ALE remap between the windows re-grids h_layer under the per-layer leftover; the console heat/salt budget shows the in-transit content as a residual until the next drain; the max(0,vol) / vanishing-layer hatch in drain_reconstruct_hprev stays non-conservative when the availability limiter does not converge.

Per-pass re-reconstruction (V2): the PPM parabola is rebuilt from the CURRENT Tr at the start of every sub-cycle pass. The hup/hlos/min_h two-test limiter sets the drained transport uhh per pass; the swept-average parabola sets the concentration multiplying it — the two are orthogonal.

Decomposition: every field the passes read off-tile is exchanged through rdb_ocean_halo (then the single-rank periodic/fold wrap), and the pass count is the global max of the tile CFL. Fields whose ghost band is a pure local function of already-exchanged inputs (tr_work, parabola coefficients, limited per-pass transport uhh, availability scale) skip the exchange (skip_halo).

   subroutine continuity_tracer_drain(grid, metrics, this, ms, ratio, bc, post_weight)
      !! Phase-2 (6b) windowed horizontal tracer-advection drain.
      !!
      !! Spends the accumulated face transports `this%uhtr/vhtr` (built
      !! over `ratio` outer steps in `TR_MODE_ACCUMULATE`) onto the
      !! frozen tracer mass `ms%tracers(:)%c` using a fixed-budget
      !! Colella-Woodward swept-average PPM sub-cycle (Adcroft &
      !! Hallberg 2006; MOM6 ADVECT_PPM).  Conservation is structural:
      !!
      !!   hprev = areaT·h_end + div(uhtr)        (≈ window-start thickness)
      !!   Tr_start = c (held)                  (window-start concentration)
      !!
      !! the drain evolves hprev → h_end while moving tracer with each
      !! limited transport portion; at the end hprev == h_end so
      !! Σ(Tr·hprev) = Σ(Tr_start·hprev_start) to round-off.  The
      !! concentration is the held `c`, which is the window-start value.
      !!
      !! Pass cap: the sub-cycle runs at most `2*ratio+1` passes, which can
      !! leave transport undrained.  That leftover is carried into the next
      !! window (`uhtr/vhtr` seeded from it, `c` untouched) and counted in
      !! `drain_carry_windows` / `drain_carry_faces`.  Known gaps: the carry
      !! is not in the restart file (a restart taken while a carry is pending
      !! drops it); an ALE remap between the windows re-grids `h_layer` under
      !! the per-layer leftover; the console heat/salt budget shows the
      !! in-transit content as a residual until the next drain; the
      !! `max(0,vol)` / vanishing-layer hatch in `drain_reconstruct_hprev`
      !! stays non-conservative when the availability limiter does not
      !! converge.
      !!
      !! Per-pass re-reconstruction (V2): the PPM parabola is rebuilt from
      !! the CURRENT Tr at the start of every sub-cycle pass.  The
      !! `hup/hlos/min_h` two-test limiter sets the drained transport
      !! `uhh` per pass; the swept-average parabola sets the concentration
      !! multiplying it — the two are orthogonal.
      !!
      !! Decomposition: every field the passes read off-tile is exchanged
      !! through `rdb_ocean_halo` (then the single-rank periodic/fold wrap),
      !! and the pass count is the global max of the tile CFL.  Fields whose
      !! ghost band is a pure local function of already-exchanged inputs
      !! (`tr_work`, parabola coefficients, limited per-pass transport
      !! `uhh`, availability scale) skip the exchange (`skip_halo`).
      type(hgrid_t), intent(in) :: grid
      type(ocean_metrics_t), intent(in) :: metrics
      type(continuity_t), intent(inout) :: this
      type(multilayer_state_t), intent(inout) :: ms
      integer, intent(in) :: ratio
         !! `dt_tracer_advect_ratio` for this window; sets the
         !! fixed-budget pass count `max_iter = 2·ratio+1` (ratio is an
         !! integer ⇒ ceil(ratio) = ratio).
      type(ocean_bc_state_t), intent(in), optional :: bc
      real(wp), intent(in), optional :: post_weight
         !! Budget pre-multiplier for the post-average writes (un-hold +
         !! drain sub-cycle): `1/ocean_budget_stage_weight(is_pc)`, i.e. 2.0
         !! under `ssp_rk2` (absent default) and 1.0 under `pred_corr`.

      integer :: nx, ny, nz, nx_phys, ny_phys, nghost
      real(wp) :: post_w
      integer :: max_iter, ipass, it
      integer(int64) :: ncarry
      integer :: i, j, k
      real(wp) :: cfl_max, cfl_loc, denom, fmax
      logical :: per_x, per_y, fold_n
      logical :: open_w, open_e, open_s, open_n

      if (.not. allocated(ms%tracers)) return

      post_w = DRAIN_BUDGET_POST_AVERAGE_WEIGHT
      if (present(post_weight)) post_w = post_weight

      nx = grid%nx_total
      ny = grid%ny_total
      nz = ms%nz_ml
      nx_phys = grid%nx_phys
      ny_phys = grid%ny_phys
      nghost = grid%nghost

      per_x = .false.
      per_y = .false.
      fold_n = .false.
      if (present(bc)) then
         per_x = bc%periodic_x .and. .not. ocean_halo_is_decomposed_x()
         per_y = bc%periodic_y .and. .not. ocean_halo_is_decomposed_y()
         fold_n = bc%north_fold
      end if
      ! WENO rung ladder: a side that continues into more domain (periodic
      ! axis, or an inter-rank neighbour) reads the full stencil; only a
      ! physical wall side degrades.  Keyed on the GLOBAL geometry so every
      ! decomposition picks the same rung at the same face.
      open_w = .false.
      open_e = .false.
      open_s = .false.
      open_n = .false.
      if (present(bc)) then
         open_w = bc%periodic_x .or. .not. bc%has_west
         open_e = bc%periodic_x .or. .not. bc%has_east
         open_s = bc%periodic_y .or. .not. bc%has_south
         open_n = bc%periodic_y .or. .not. bc%has_north
      end if

      ! ---- Halo of the accumulators (single-rank seams) ----
      ! hprev reads neighbour uhtr/vhtr across the ghost, so wrap them
      ! first (inter-rank face halo when decomposed, else the seam wrap).
      call drain_wrap_face_x(this%uhtr, nx, ny, nz, nx_phys, ny_phys, nghost, &
                             per_x, per_y, fold_n)
      call drain_wrap_face_y(this%vhtr, nx, ny, nz, nx_phys, ny_phys, nghost, &
                             per_x, per_y, fold_n)

      ! ---- Conservative availability limiter (Fox-Kemper × windowed-advect) ----
      ! Tighten the accumulated window transports so every cell's
      ! reconstructed window-start volume stays ≥ areaT·h_min — the
      ! combined FK + resolved transport can otherwise overdraw a thin
      ! z* surface layer (vol < 0), tripping the non-conservative
      ! clamp/hatch below and leaking ~5%/day of tracer.  Limited uhtr/vhtr
      ! feed BOTH the reconstruction AND the sub-cycle seed (drain_copy_3d
      ! below), so the drain still telescopes exactly onto h_end and
      ! Σ(areaT·h·c) is conserved.  Inert (bit-identical) when no cell
      ! would otherwise go vol ≤ areaT·h_min (ratio=1 / FK off / benign
      ! windows).  tr_work is reused as the per-cell scale scratch — it is
      ! overwritten by drain_parabola_* in the sub-cycle before any read.
      ! Fixed-budget FCT tighten (GPU-uniform, no data-dependent exit);
      ! AVAIL_LIMIT_PASS bounds the constraint diffusion across the stencil.
      do ipass = 1, AVAIL_LIMIT_PASS
         call drain_avail_limit(nx, ny, nz, metrics%areaT, this%h_min, &
                                ms%h_layer, this%uhtr, this%vhtr, this%tr_work)
         call drain_wrap_centre(this%tr_work, nx, ny, nz, nx_phys, ny_phys, &
                                nghost, per_x, per_y, fold_n)
         call drain_avail_scale_x(nx, ny, nz, this%tr_work, this%uhtr)
         call drain_avail_scale_y(nx, ny, nz, this%tr_work, this%vhtr)
         call drain_wrap_face_x(this%uhtr, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n)
         call drain_wrap_face_y(this%vhtr, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n)
      end do

      ! ---- Reconstruct hprev = areaT·h_end + div(uhtr) (volume → thickness) ----
      ! + vanishing-layer hatch.  Done on the FROZEN window-start grid: the cell's
      ! window-start thickness, against which the held concentration is
      ! consistent.
      call drain_reconstruct_hprev(nx, ny, nz, metrics%areaT, metrics%iareaT, &
                                   ms%h_layer, this%uhtr, this%vhtr, this%hprev_work)
      call drain_wrap_centre(this%hprev_work, nx, ny, nz, nx_phys, ny_phys, nghost, &
                             per_x, per_y, fold_n)

      ! Release the per-stage budget hold booked by
      ! `continuity_tracer_step_split`: `c` is the held window-start
      ! concentration and the sub-cycle below weights it by the window-START
      ! thickness `hprev` it just reconstructed, so the `Σ c·δh` booked per
      ! stage against the evolving `h_layer` is cancelled here.
      if (this%window_hold_open) then
         call tracer_window_release(this, ms, nx, ny, nz, post_w)
         this%window_hold_open = .false.
      end if

      ! ---- Seam-wrap the held tracer before the first reconstruction ----
      ! The PPM parabola (drain_parabola_*) reads the seam-adjacent cells'
      ! tracer concentration Tr = c at the ghost band (a ±2 stencil).
      ! hprev is already periodic/fold-consistent (wrapped above), but the
      ! held c the drain inherits is NOT guaranteed periodic in its ghost
      ! halo (the dynamics wrap h_layer, not necessarily the windowed c), so
      ! a seam-physical cell's reconstruction read a different Tr than its
      ! translated interior image — breaking bit-exact translation invariance
      ! (the per-pass updates re-wrap c at the end, but the FIRST pass's
      ! parabola already consumed the stale ghosts).  Wrap every advected
      ! tracer's c once here so pass 1 sees the same periodic/fold ghosts
      ! the later passes do.  No-op on non-periodic walls (bit-identical).
      do it = 1, size(ms%tracers)
         if (.not. allocated(ms%tracers(it)%c)) cycle
         if (.not. ms%tracers(it)%do_horizontal_advection) cycle
         call drain_wrap_centre(ms%tracers(it)%c, nx, ny, nz, nx_phys, ny_phys, &
                                nghost, per_x, per_y, fold_n)
      end do

      ! ---- Seed remaining transport from the accumulators ----
      call drain_copy_3d(nx + 1, ny, nz, this%uhtr, this%uhr_x)
      call drain_copy_3d(nx, ny + 1, nz, this%vhtr, this%uhr_y)

      ! ---- Fixed-budget sub-cycle, budget sized to the ACTUAL courant ----
      ! The worst case is `2*ratio+1` (per-step CFL ~ 1), but a real run's
      ! accumulated tracer courant is much smaller, so most of those passes are
      ! exact no-ops (once a face's transport is drained the hup/hlos limiter
      ! yields uhh=0 and the pass changes nothing).  Reduce the max per-cell
      ! accumulated courant ONCE (grid-uniform — every column runs the same
      ! `max_iter`, so no warp divergence; no per-pass early-exit) and size the
      ! fixed loop with the SAME `2*ceil(C)+1` form, capped at the worst case.
      ! Dropping the no-op passes is BIT-IDENTICAL (verified vs `2*ratio+1`).
      cfl_max = 0.0_wp
      do concurrent(k=1:nz, j=1:ny, i=1:nx) reduce(max:cfl_max)
         denom = metrics%areaT(i, j)*max(this%hprev_work(i, j, k), this%h_min)
         fmax = max(abs(this%uhr_x(i, j, k)), abs(this%uhr_x(i + 1, j, k)), &
                    abs(this%uhr_y(i, j, k)), abs(this%uhr_y(i, j + 1, k)))
         cfl_max = max(cfl_max, fmax/denom)
      end do
      ! Passes needed = ceiling(cfl_max / per-pass capacity).  The drain_limit_x
      ! two-test limiter guarantees each pass moves at least 0.5·hup (the
      ! `max(0.5*hup, hup-hlos)` floor), i.e. worst-case capacity 0.5 CFL/pass,
      ! so ceiling(2·cfl_max) passes are provably sufficient (and exact in the
      ! divergent worst case).  This is much tighter than the old
      ! `2*ceiling(cfl_max)+1`: at the default ratio = 1 a stable step has
      ! cfl < 0.5 ⇒ ONE pass (was 3 — two full-grid no-op passes).  Still
      ! grid-uniform (one reduction, no per-pass divergence) and capped at the
      ! worst case `2*ratio+1`.
      ! Decomposed: every rank must run the SAME pass count (each pass holds
      ! collective halo exchanges), so take the global max of the tile CFL.
      ! A max is exact in FP, hence layout-independent.  Identity on one rank.
      if (ocean_halo_is_decomposed_x() .or. ocean_halo_is_decomposed_y()) then
         cfl_loc = cfl_max
         call halo_allreduce_max(cfl_loc, cfl_max)
      end if
      max_iter = ceiling(2.0_wp*cfl_max)
      if (max_iter > 2*ratio + 1) max_iter = 2*ratio + 1
      if (max_iter < 1) max_iter = 1
      this%drain_iter_max = max(this%drain_iter_max, max_iter)
      do ipass = 1, max_iter
         ! Zonal sub-pass --------------------------------------------------
         call drain_limit_x(nx, ny, nz, metrics%areaT, this%h_min, &
                            this%uhr_x, this%hprev_work, this%uhh_x)
         ! Wrap the limited per-pass transport so the swept flux at the
         ! periodic seam / north fold reads the SAME uhh from both sides of
         ! the wrapped face (drain_limit_x writes only interior faces 2..nx
         ! + zeros the array edges; without the wrap the two images of a
         ! seam face disagree ⇒ the flux divergence does not telescope and
         ! Σ(areaT·h·c) leaks at the seam).  No-op on non-periodic walls.
         call drain_wrap_face_x(this%uhh_x, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n, skip_halo=.true.)
         do it = 1, size(ms%tracers)
            if (.not. allocated(ms%tracers(it)%c)) cycle
            if (.not. ms%tracers(it)%do_horizontal_advection) cycle
            if (this%tracer_h3) then
               ! ppm_h3: Huynh edges + CW84 limiter on Tr = c; uhh is
               ! a volume so the flux below is a tracer volume.
               call drain_fill_conc(nx, ny, nz, ms%tracers(it)%c, this%hprev_work, this%tr_work)
               call drain_wrap_centre(this%tr_work, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, skip_halo=.true.)
               call drain_swept_flux_x_h3(nx, ny, nz, metrics%areaT, this%uhh_x, &
                                          this%hprev_work, this%tr_work, metrics%wet_T, &
                                          this%tr_flux_x)
            else if (this%tracer_recon == TRACER_RECON_PPM) then
               call drain_parabola_x(nx, ny, nz, metrics%wet_T, ms%tracers(it)%c, this%hprev_work, &
                                     this%tr_work, this%pal, this%par, this%pa6)
               ! Wrap the parabola coefficients so a seam-face donor that lands
               ! in a ghost column carries the SAME (full-PPM) reconstruction as
               ! its physical image — the ghost band is otherwise PCM, which
               ! makes the two seam-face flux evaluations disagree (seam leak).
               ! Batch the 3 independent parabola-coeff wraps async on queue 1,
               ! sync once before the swept flux reads them.
               call drain_wrap_centre(this%pal, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, no_wait=.true., skip_halo=.true.)
               call drain_wrap_centre(this%par, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, no_wait=.true., skip_halo=.true.)
               call drain_wrap_centre(this%pa6, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, no_wait=.true., skip_halo=.true.)
               !$acc wait(1)
               call drain_swept_flux_x(nx, ny, nz, metrics%areaT, this%uhh_x, &
                                       this%hprev_work, this%pal, this%par, this%pa6, &
                                       this%tr_flux_x)
            else
               ! Q6 WENO drain: reconstruct the swept-average donor-edge
               ! concentration with the WENO ladder instead of the CW
               ! parabola.  Build Tr = c, wrap it so seam-adjacent
               ! stencils read consistent ghosts (the WENO reach is wider
               ! than PPM's — the wrap must cover the widest rung), then
               ! evaluate the swept face flux directly from Tr.
               call drain_fill_conc(nx, ny, nz, ms%tracers(it)%c, this%hprev_work, this%tr_work)
               call drain_wrap_centre(this%tr_work, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, skip_halo=.true.)
               call drain_swept_flux_x_weno(nx, ny, nz, nghost, open_w, open_e, metrics%areaT, &
                                            this%uhh_x, this%hprev_work, this%tr_work, &
                                            metrics%wet_T, this%tracer_recon, this%tr_flux_x)
            end if
            select case (ms%tracers(it)%budget_id)
            case (TRACER_BUDGET_HEAT)
               call drain_update_tracer_x_budget(nx, ny, nz, metrics%iareaT, this%tr_flux_x, &
                                                 this%uhh_x, this%hprev_work, post_w, &
                                                 ms%tracers(it)%c, ms%heat_budget_horiz_adv)
            case (TRACER_BUDGET_SALT)
               call drain_update_tracer_x_budget(nx, ny, nz, metrics%iareaT, this%tr_flux_x, &
                                                 this%uhh_x, this%hprev_work, post_w, &
                                                 ms%tracers(it)%c, ms%salt_budget_horiz_adv)
            case default
               call drain_update_tracer_x(nx, ny, nz, metrics%iareaT, this%tr_flux_x, &
                                          this%uhh_x, this%hprev_work, ms%tracers(it)%c)
            end select
            call drain_wrap_centre(ms%tracers(it)%c, nx, ny, nz, nx_phys, ny_phys, &
                                   nghost, per_x, per_y, fold_n)
         end do
         ! Advance the (shared) thickness + remaining transport once.
         call drain_update_h_x(nx, ny, nz, metrics%iareaT, this%uhh_x, this%hprev_work)
         call drain_subtract_3d(nx + 1, ny, nz, this%uhh_x, this%uhr_x)
         call drain_wrap_centre(this%hprev_work, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n)
         call drain_wrap_face_x(this%uhr_x, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n)

         ! Meridional sub-pass ---------------------------------------------
         call drain_limit_y(nx, ny, nz, metrics%areaT, this%h_min, &
                            this%uhr_y, this%hprev_work, this%uhh_y)
         ! Wrap the limited meridional transport (see the zonal note above).
         call drain_wrap_face_y(this%uhh_y, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n, skip_halo=.true.)
         do it = 1, size(ms%tracers)
            if (.not. allocated(ms%tracers(it)%c)) cycle
            if (.not. ms%tracers(it)%do_horizontal_advection) cycle
            if (this%tracer_h3) then
               ! ppm_h3: Huynh edges + CW84 limiter on Tr = c; uhh is
               ! a volume so the flux below is a tracer volume.
               call drain_fill_conc(nx, ny, nz, ms%tracers(it)%c, this%hprev_work, this%tr_work)
               call drain_wrap_centre(this%tr_work, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, skip_halo=.true.)
               call drain_swept_flux_y_h3(nx, ny, nz, metrics%areaT, this%uhh_y, &
                                          this%hprev_work, this%tr_work, metrics%wet_T, &
                                          this%tr_flux_y)
            else if (this%tracer_recon == TRACER_RECON_PPM) then
               call drain_parabola_y(nx, ny, nz, metrics%wet_T, ms%tracers(it)%c, this%hprev_work, &
                                     this%tr_work, this%pal, this%par, this%pa6)
               ! Wrap the meridional parabola coefficients (see the zonal note).
               ! Batch the 3 independent parabola-coeff wraps async on queue 1,
               ! sync once before the swept flux reads them.
               call drain_wrap_centre(this%pal, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, no_wait=.true., skip_halo=.true.)
               call drain_wrap_centre(this%par, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, no_wait=.true., skip_halo=.true.)
               call drain_wrap_centre(this%pa6, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, no_wait=.true., skip_halo=.true.)
               !$acc wait(1)
               call drain_swept_flux_y(nx, ny, nz, metrics%areaT, this%uhh_y, &
                                       this%hprev_work, this%pal, this%par, this%pa6, &
                                       this%tr_flux_y)
            else
               ! Q6 WENO drain (meridional; see the zonal branch note).
               call drain_fill_conc(nx, ny, nz, ms%tracers(it)%c, this%hprev_work, this%tr_work)
               call drain_wrap_centre(this%tr_work, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                      per_x, per_y, fold_n, skip_halo=.true.)
               call drain_swept_flux_y_weno(nx, ny, nz, nghost, open_s, open_n, metrics%areaT, &
                                            this%uhh_y, this%hprev_work, this%tr_work, &
                                            metrics%wet_T, this%tracer_recon, this%tr_flux_y)
            end if
            select case (ms%tracers(it)%budget_id)
            case (TRACER_BUDGET_HEAT)
               call drain_update_tracer_y_budget(nx, ny, nz, metrics%iareaT, this%tr_flux_y, &
                                                 this%uhh_y, this%hprev_work, post_w, &
                                                 ms%tracers(it)%c, ms%heat_budget_horiz_adv)
            case (TRACER_BUDGET_SALT)
               call drain_update_tracer_y_budget(nx, ny, nz, metrics%iareaT, this%tr_flux_y, &
                                                 this%uhh_y, this%hprev_work, post_w, &
                                                 ms%tracers(it)%c, ms%salt_budget_horiz_adv)
            case default
               call drain_update_tracer_y(nx, ny, nz, metrics%iareaT, this%tr_flux_y, &
                                          this%uhh_y, this%hprev_work, ms%tracers(it)%c)
            end select
            call drain_wrap_centre(ms%tracers(it)%c, nx, ny, nz, nx_phys, ny_phys, &
                                   nghost, per_x, per_y, fold_n)
         end do
         call drain_update_h_y(nx, ny, nz, metrics%iareaT, this%uhh_y, this%hprev_work)
         call drain_subtract_3d(nx, ny + 1, nz, this%uhh_y, this%uhr_y)
         call drain_wrap_centre(this%hprev_work, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n)
         call drain_wrap_face_y(this%uhr_y, nx, ny, nz, nx_phys, ny_phys, nghost, &
                                per_x, per_y, fold_n)
      end do

      ! ---- Carry undrained transport into the next window ----
      ! The `2*ratio+1` pass cap can stop the sub-cycle with transport left in
      ! `uhr_x/uhr_y` (and then `hprev_work /= h_layer`).  That leftover is
      ! CARRIED: `uhtr/vhtr` are seeded from it, the next window's accumulate
      ! stages add on top, and the next drain's `drain_reconstruct_hprev`
      ! rebuilds the thickness `c` was left paired with.  `c` is not touched,
      ! so a uniform tracer stays uniform and content Sum(areaT*hprev*c) is
      ! conserved across windows (the residual c*div(leftover) is in transit
      ! until that drain; Adcroft & Hallberg 2006; Griffies et al. 2020).
      ! A fully drained window leaves exactly 0.0 (the limiter passes the whole
      ! remaining transport, so uhr - uhh == 0), i.e. the old zero-reset.
      ! Known gaps: the carry is not in the restart file; an ALE remap between
      ! the windows re-grids h_layer under the per-layer leftover; the console
      ! heat/salt budget shows the in-transit content as a residual until the
      ! next drain; and the max(0,vol) / vanishing-layer hatch in
      ! `drain_reconstruct_hprev` stays non-conservative when the availability
      ! limiter does not converge (unchanged).
      call drain_carry_transport(nx, ny, nz, this%uhr_x, this%uhr_y, this%uhtr, this%vhtr, &
                                 ncarry)
      if (ncarry > 0_int64) then
         this%drain_carry_windows = this%drain_carry_windows + 1
         this%drain_carry_faces = this%drain_carry_faces + ncarry
      end if
      ! ---- Reset the window clock for the next window ----
      this%t_dyn_rel_adv = 0.0_wp
   end subroutine continuity_tracer_drain

tracer_advect_zonal_one_impl

private pure subroutine tracer_advect_zonal_one_impl(nx, ny, nz, dt, iareaT, wet_T, h_before, h_after, c, mass_flux_x, Tr_face_left_x, Tr_face_right_x, budget_adv, budget_w, h3, h_floor)

Zonal half of tracer_advect_one_impl. Same three-pass pattern (PPM reconstruction → upwind pick into tracer mass flux → forward-Euler update) but only the x-direction half. Reads the input h for the Tr = c/h reconstruction. The tracer transport mass_flux_x·Tr_face inherits dy_cu, and the divergence closes with iareaT (= inv_dx on uniform). Mirror-T at land neighbours (C2): a held land column’s tracer is reflected to the local cell so the wet-side face value is unbiased; bit-identical for all-wet (wet_T≡1).

Optional budget_adv: when present, a separate do concurrent loop accumulates −dt·(Tr_face_left_x(i+1)−Tr_face_left_x(i))·iareaT into budget_adv (same divergence written to c). The prognostic c update loop is UNCHANGED — byte-identical when budget_adv is absent. Used by the console salt/heat closure.

   pure subroutine tracer_advect_zonal_one_impl(nx, ny, nz, dt, iareaT, wet_T, h_before, h_after, c, &
                                                mass_flux_x, &
                                                Tr_face_left_x, Tr_face_right_x, &
                                                budget_adv, budget_w, h3, h_floor)
      !! Zonal half of `tracer_advect_one_impl`.  Same three-pass
      !! pattern (PPM reconstruction → upwind pick into tracer mass
      !! flux → forward-Euler update) but only the x-direction half.
      !! Reads the input `h` for the Tr = c/h reconstruction.  The
      !! tracer transport `mass_flux_x·Tr_face` inherits `dy_cu`, and
      !! the divergence closes with `iareaT` (= `inv_dx` on uniform).
      !! Mirror-T at land neighbours (C2): a held land column's tracer
      !! is reflected to the local cell so the wet-side face value is
      !! unbiased; bit-identical for all-wet (`wet_T≡1`).
      !!
      !! Optional `budget_adv`: when present, a separate `do concurrent`
      !! loop accumulates `−dt·(Tr_face_left_x(i+1)−Tr_face_left_x(i))·iareaT`
      !! into `budget_adv` (same divergence written to `c`).  The
      !! prognostic c update loop is UNCHANGED — byte-identical when
      !! `budget_adv` is absent.  Used by the console salt/heat closure.
      integer, intent(in) :: nx, ny, nz
      real(wp), intent(in) :: dt
      real(wp), intent(in) :: iareaT(nx, ny)
      real(wp), intent(in) :: wet_T(nx, ny)
      real(wp), intent(in) :: h_before(nx, ny, nz)
      real(wp), intent(in) :: h_after(nx, ny, nz)
         !! The thickness continuity wrote for THIS direction (the `h⁺` of the
         !! `h⁻ → h⁺` pair; `h_before` is the `h⁻`).  Retained for the pair's
         !! contract; the update divides by the UNFLOORED `h⁺` recomputed from
         !! the fluxes (see the update loop).
      real(wp), intent(inout) :: c(nx, ny, nz)
      real(wp), intent(in) :: mass_flux_x(nx + 1, ny, nz)
      real(wp), intent(inout) :: Tr_face_left_x(nx + 1, ny, nz)
      real(wp), intent(inout) :: Tr_face_right_x(nx + 1, ny, nz)
      real(wp), intent(inout), optional :: budget_adv(nx, ny, nz)
         !! Per-cell accumulator for the horizontal-advection budget
         !! (same sign/units as c).  When present, the zonal-flux
         !! divergence is added (+=) here after the prognostic update.
         !! Absent ⇒ inert (byte-identical to the pre-feature build).
      real(wp), intent(in), optional :: budget_w
         !! Bookkeeping weight on the `budget_adv` increment (absent ⇒ 1,
         !! bit-identical).  A caller writing OUTSIDE an RK2 stage (the GM
         !! operator, after the stage average) passes the reciprocal of the
         !! console's per-step weight, `1/ocean_budget_stage_weight`.
      logical, intent(in), optional :: h3
         !! `.true.` selects the `ppm_h3` flux (see `ppm_h3_face_flux`).
         !! Absent or `.false.` ⇒ the PPM path below, textually unchanged.
      real(wp), intent(in), optional :: h_floor
         !! Angstrom floor (m) the paired continuity stored as `max(h⁺, h_floor)`.
         !! > 0 ⇒ where the floor binds the update divides by the floored
         !! thickness and dilutes c, so Sum(h·c) is unchanged by the floor
         !! (KEEP-CONTENT semantics; the injected water carries no tracer).
         !! Absent or 0 ⇒ the loop below is textually unchanged.

      integer :: i, j, k
      real(wp) :: div_m, div_f, h_new, hf
      logical :: use_h3
      real(wp) :: Tr_m2, Tr_m1, Tr_0, Tr_p1, Tr_p2
      real(wp) :: dh_m1, dh_0, dh_p1, Tr_left, Tr_right
      real(wp) :: mass_x, Tr_face, dt_b, hc, wm
      integer :: ic

      hf = 0.0_wp
      if (present(h_floor)) hf = h_floor
      use_h3 = .false.
      if (present(h3)) use_h3 = h3

      if (use_h3) then
         ! ppm_h3: Huynh 3rd-order edges, CW84 limiter, swept-volume flux.
         ! Interior faces need cells i-2..i+1; the two faces next to each
         ! wall fall back to the first-order upwind cell value.
         do concurrent(k=1:nz, j=1:ny, i=3:nx - 1) &
            local(Tr_m1, Tr_0, Tr_p1, mass_x, hc, wm, ic)
            mass_x = mass_flux_x(i, j, k)
            if (mass_x >= 0.0_wp) then
               ic = i - 1
            else
               ic = i
            end if
            hc = h_before(ic, j, k)
            Tr_0 = c(ic, j, k)
            Tr_m1 = ppm_mirror_h(c(ic - 1, j, k), Tr_0, wet_T(ic - 1, j))
            Tr_p1 = ppm_mirror_h(c(ic + 1, j, k), Tr_0, wet_T(ic + 1, j))
            wm = wet_T(ic - 1, j)*wet_T(ic, j)*wet_T(ic + 1, j)
            Tr_face_left_x(i, j, k) = ppm_h3_face_flux(Tr_m1, Tr_0, Tr_p1, wm, &
                                                       mass_x, abs(mass_x)*dt*iareaT(ic, j)/hc)
         end do
         ! Upwind faces next to each wall: 1:2 and nx:nx+1, disjoint from the
         ! interior 3:nx-1 (and together the whole 1:nx+1 when nx <= 3).
         do concurrent(k=1:nz, j=1:ny, i=1:min(2, nx + 1)) local(mass_x, ic)
            mass_x = mass_flux_x(i, j, k)
            if (mass_x >= 0.0_wp) then
               ic = i - 1
            else
               ic = i
            end if
            ic = max(1, min(nx, ic))
            Tr_face_left_x(i, j, k) = mass_x*(c(ic, j, k))
         end do
         do concurrent(k=1:nz, j=1:ny, i=max(3, nx):nx + 1) local(mass_x, ic)
            mass_x = mass_flux_x(i, j, k)
            if (mass_x >= 0.0_wp) then
               ic = i - 1
            else
               ic = i
            end if
            ic = max(1, min(nx, ic))
            Tr_face_left_x(i, j, k) = mass_x*(c(ic, j, k))
         end do
      else
         do concurrent(k=1:nz, j=1:ny, i=3:nx - 2) &
            local(Tr_m2, Tr_m1, Tr_0, Tr_p1, Tr_p2, &
                  dh_m1, dh_0, dh_p1, Tr_left, Tr_right)
            Tr_0 = c(i, j, k)
            Tr_m1 = ppm_mirror_h(c(i - 1, j, k), Tr_0, wet_T(i - 1, j))
            Tr_p1 = ppm_mirror_h(c(i + 1, j, k), Tr_0, wet_T(i + 1, j))
            Tr_m2 = ppm_mirror_h(c(i - 2, j, k), Tr_m1, wet_T(i - 2, j))
            Tr_p2 = ppm_mirror_h(c(i + 2, j, k), Tr_p1, wet_T(i + 2, j))
            call ppm_limited_slope(Tr_m2, Tr_m1, Tr_0, dh_m1)
            call ppm_limited_slope(Tr_m1, Tr_0, Tr_p1, dh_0)
            call ppm_limited_slope(Tr_0, Tr_p1, Tr_p2, dh_p1)
            dh_0 = dh_0*wet_T(i - 1, j)*wet_T(i, j)*wet_T(i + 1, j)
            Tr_left = 0.5_wp*(Tr_m1 + Tr_0) - (dh_0 - dh_m1)/6.0_wp
            Tr_right = 0.5_wp*(Tr_0 + Tr_p1) - (dh_p1 - dh_0)/6.0_wp
            call ppm_cell_limiter(Tr_0, Tr_left, Tr_right)
            Tr_face_right_x(i, j, k) = Tr_left
            Tr_face_left_x(i + 1, j, k) = Tr_right
         end do
         do concurrent(k=1:nz, j=1:ny)
            Tr_face_left_x(1, j, k) = c(1, j, k)
            Tr_face_right_x(1, j, k) = c(1, j, k)
            Tr_face_left_x(2, j, k) = c(1, j, k)
            Tr_face_right_x(2, j, k) = c(2, j, k)
            Tr_face_left_x(3, j, k) = c(2, j, k)
            Tr_face_right_x(3, j, k) = c(2, j, k)
            Tr_face_right_x(nx - 1, j, k) = c(nx - 1, j, k)
            Tr_face_left_x(nx, j, k) = c(nx - 1, j, k)
            Tr_face_right_x(nx, j, k) = c(nx, j, k)
            Tr_face_left_x(nx + 1, j, k) = c(nx, j, k)
            Tr_face_right_x(nx + 1, j, k) = c(nx, j, k)
         end do

         do concurrent(k=1:nz, j=1:ny, i=1:nx + 1) local(mass_x, Tr_face)
            mass_x = mass_flux_x(i, j, k)
            if (mass_x >= 0.0_wp) then
               Tr_face = Tr_face_left_x(i, j, k)
            else
               Tr_face = Tr_face_right_x(i, j, k)
            end if
            Tr_face_left_x(i, j, k) = mass_x*Tr_face
         end do
      end if

      ! Concentration update, c+ = c + dt*(c*div(Phi) - div(F))/h+, algebraically
      ! (c h- - dt div F)/h+ with h+ = h- - dt div(Phi) (Griffies et al. 2020;
      ! Adcroft & Hallberg 2006).  h+ is the UNFLOORED thickness continuity
      ! computes from the same fluxes, so a uniform c is reproduced to round-off
      ! whatever h does; a layer left empty (h+ <= H_DIV_EPS) keeps its c.
      ! With an Angstrom floor (hf > 0) that binds, continuity STORES hf > h+:
      ! c is then formed against the stored hf, (h- c - dt div F)/hf, so the
      ! floor dilutes c and leaves the content h*c alone (no tracer is created
      ! with the injected water).  An overdrawn cell (h+ <= H_DIV_EPS) is not
      ! physical content: it keeps c, so the floor creates at most hf*c there.
      do concurrent(k=1:nz, j=1:ny, i=1:nx) local(div_m, div_f, h_new)
         div_m = (mass_flux_x(i + 1, j, k) - mass_flux_x(i, j, k))*iareaT(i, j)
         h_new = h_before(i, j, k) - dt*div_m
         if (hf > 0.0_wp .and. h_new > H_DIV_EPS .and. h_new < hf) then
            ! Angstrom floor binds: continuity stored hf, not h_new.  Divide by
            ! the stored thickness so h*c is unchanged by the floor.  Only for
            ! H_DIV_EPS < h_new < hf: an overdrawn cell (h_new <= H_DIV_EPS)
            ! is not physical content (c*h_new/hf would blow c up as h_new
            ! goes negative), so it falls through, c is kept (bounded), and the
            ! floor then creates at most hf*c per floored cell.
            div_f = (Tr_face_left_x(i + 1, j, k) - Tr_face_left_x(i, j, k))*iareaT(i, j)
            c(i, j, k) = c(i, j, k) + (c(i, j, k)*(h_before(i, j, k) - hf) - dt*div_f)/hf
         else if (h_new > H_DIV_EPS) then
            div_f = (Tr_face_left_x(i + 1, j, k) - Tr_face_left_x(i, j, k))*iareaT(i, j)
            c(i, j, k) = c(i, j, k) + dt*(c(i, j, k)*div_m - div_f)/h_new
         end if
      end do

      ! Budget accumulator: separate guarded loop so the prognostic update above
      ! is UNCHANGED (byte-identical when budget_adv is absent).  Mirrors the
      ! same divergence that was just applied to c — the console uses it to
      ! close the salt/heat conservation residual.
      if (present(budget_adv)) then
         dt_b = dt
         if (present(budget_w)) dt_b = dt*budget_w
         do concurrent(k=1:nz, j=1:ny, i=1:nx)
            budget_adv(i, j, k) = budget_adv(i, j, k) - dt_b* &
                                  (Tr_face_left_x(i + 1, j, k) - Tr_face_left_x(i, j, k))*iareaT(i, j)
         end do
      end if
   end subroutine tracer_advect_zonal_one_impl

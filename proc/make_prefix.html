   function make_prefix(path, line) result(p)
      !! Build a `path:line: ` error prefix.
      character(len=*), intent(in) :: path
      integer, intent(in) :: line
      character(len=:), allocatable :: p
      p = path//":"//fmt_int(line)//": "
   end function make_prefix

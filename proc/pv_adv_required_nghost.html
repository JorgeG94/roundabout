public pure function pv_adv_required_nghost(code) result(ng)

Minimum nghost for a PV face-interp scheme’s stencil radius: weno5 (radius 3) -> 3, weno7 (radius 4) -> 4; centered/weno3 fit the nghost>=2 baseline. Mirrors the tracer-WENO ladder’s per-rung gate; configure fail-loud rejects an under-provisioned halo.

   pure function pv_adv_required_nghost(code) result(ng)
      !! Minimum `nghost` for a PV face-interp scheme's stencil radius:
      !! weno5 (radius 3) -> 3, weno7 (radius 4) -> 4; centered/weno3 fit the
      !! nghost>=2 baseline.  Mirrors the tracer-WENO ladder's per-rung gate;
      !! `configure` fail-loud rejects an under-provisioned halo.
      integer, intent(in) :: code
      integer :: ng
      select case (code)
      case (PV_ADV_WENO5)
         ng = 3
      case (PV_ADV_WENO7)
         ng = 4
      case default
         ng = 2
      end select
   end function pv_adv_required_nghost

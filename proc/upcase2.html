   pure function upcase2(s) result(u)
      !! Uppercase the first two characters of a token.
      character(len=*), intent(in) :: s
      character(len=2) :: u
      integer :: k, ic
      u = "  "
      do k = 1, min(2, len_trim(s))
         ic = iachar(s(k:k))
         if (ic >= iachar("a") .and. ic <= iachar("z")) ic = ic - 32
         u(k:k) = achar(ic)
      end do
   end function upcase2

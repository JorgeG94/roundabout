   subroutine scratch_3d_buffer_destroy(this)
      class(scratch_3d_buffer_t), intent(inout) :: this
      this%is_init = .false.
      if (allocated(this%data)) deallocate (this%data)
      this%n1 = 0
      this%n2 = 0
      this%n3 = 0
      this%name = ""
   end subroutine scratch_3d_buffer_destroy

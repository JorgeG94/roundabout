   pure function ns_rank_north() result(r)
      !! North neighbour rank with periodic wrap-around.
      integer :: r
      integer :: ry_n
      ry_n = oh_decomp%ry + 1
      if (ry_n >= oh_decomp%py) ry_n = 0
      r = decomp_rank_from_coords(oh_decomp%px, oh_decomp%rx, ry_n)
   end function ns_rank_north
